            OR = {

                        leader = {
                            is_councilor = yes

                                has_trait = leader_trait_enduring_bulwark
                            }
                        }
                    }
                }
                AND = {
                    exists = defender
                    defender = {
                        leader = {
                            is_councilor = yes
                            OR = {
                                has_trait = leader_trait_enduring_bulwark
                            }
                        }
                    }
                }
            }
        }
    }

    immediate = {
        remove_modifier = daunting_smite

                    is_councilor = yes
                    has_trait = leader_trait_enduring_bulwark
                }
            }
            add_modifier = {
                modifier = daunting_smite
                days = 360
            }
        }
    }
}
