country_event = {
    id = hsr_eco.100
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        # HSR_FINANCE_YEAR_BEGIN 年度唯一时点：先结息，再发税收，最后更新下一年度额度。
        set_variable = { which = hsr_finance_current_year value = value:hsr_finance_calendar_year }
        if = {
            limit = {
                OR = {
                    NOT = {
                        has_country_flag = hsr_finance_year_initialized
                    }
                    check_variable = { which = hsr_finance_last_settlement_year value < hsr_finance_current_year }
                }
            }
            set_country_flag = hsr_finance_year_initialized
            set_variable = { which = hsr_finance_last_settlement_year value = hsr_finance_current_year }
            hsr_finance_settle_interest = yes
            set_variable = { which = hsr_credits_country_year_income_gross value = 0 }
            every_owned_colony = {
                limit = {
                    hsr_credits_is_live_colony = yes
                }
                #地方政策
                if = {
                    limit = {
                        not = {
                            has_deposit = d_colony_credits_slot
                        }
                    }
                    add_deposit = d_colony_credits_slot
                }
                if = {
                    limit = {
                        or = {
                            check_variable = { which = hsr_planet_social value < 1 }
                            check_variable = { which = hsr_planet_social value > 3 }
                        }
                    }
                    set_variable = { which = hsr_planet_social value = 2 }
                }
                if = {
                    limit = {
                        or = {
                            check_variable = { which = hsr_planet_governance value < 1 }
                            check_variable = { which = hsr_planet_governance value > 3 }
                        }
                    }
                    set_variable = { which = hsr_planet_governance value = 2 }
                }
                if = {
                    limit = {
                        or = {
                            check_variable = { which = hsr_planet_military value < 1 }
                            check_variable = { which = hsr_planet_military value > 3 }
                        }
                    }
                    set_variable = { which = hsr_planet_military value = 2 }
                }
                if = {
                    limit = {
                        or = {
                            check_variable = { which = hsr_planet_industrial value < 1 }
                            check_variable = { which = hsr_planet_industrial value > 3 }
                        }
                    }
                    set_variable = { which = hsr_planet_industrial value = 2 }
                }
                #读取数据
                hsr_credits_read_production = yes
                #计算数据
                if = {
                    limit = {
                        check_variable = { which = hsr_credits_net value > 0 }
                    }
                    if = {
                        limit = {
                            root = {
                                has_policy_flag = hsr_basics_tax_revenue_captial
                            }
                        }
                        set_variable = {
                            which = hsr_credits_planet_year_value
                            value = value:hsr_add_mult_value|BASEVAR|0.3|VAR1|hsr_credits_net|
                        }
                        set_variable = {
                            which = hsr_credits_country_year_value
                            value = value:hsr_add_mult_value|BASEVAR|0.7|VAR1|hsr_credits_net|
                        }
                    }
                    else_if = {
                        limit = {
                            root = {
                                has_policy_flag = hsr_basics_tax_revenue_local
                            }
                        }
                        set_variable = {
                            which = hsr_credits_planet_year_value
                            value = value:hsr_add_mult_value|BASEVAR|0.7|VAR1|hsr_credits_net|
                        }
                        set_variable = {
                            which = hsr_credits_country_year_value
                            value = value:hsr_add_mult_value|BASEVAR|0.3|VAR1|hsr_credits_net|
                        }
                    }
                    else_if = {
                        limit = {
                            root = {
                                has_policy_flag = hsr_basics_tax_revenue_average
                            }
                        }
                        set_variable = {
                            which = hsr_credits_planet_year_value
                            value = value:hsr_add_mult_value|BASEVAR|0.5|VAR1|hsr_credits_net|
                        }
                        set_variable = {
                            which = hsr_credits_country_year_value
                            value = value:hsr_add_mult_value|BASEVAR|0.5|VAR1|hsr_credits_net|
                        }
                    }
                    change_variable = { which = hsr_credits_planet_trade_storage value = hsr_credits_planet_year_value }
                    prev = {
                        change_variable = {
                            which = hsr_credits_country_trade_storage
                            value = prev.hsr_credits_country_year_value
                        }
                        #总存储
                        change_variable = {
                            which = hsr_credits_country_year_income_gross
                            value = prev.hsr_credits_country_year_value
                        }
                        #年收入
                    }
                }
                else_if = {
                    limit = {
                        check_variable = { which = hsr_credits_net value < 0 }
                    }
                    change_variable = { which = hsr_credits_planet_trade_storage value = hsr_credits_net }
                }
                #建议显示
                if = {
                    limit = {
                        not = {
                            has_modifier = hsr_credits_planet_modifiers
                        }
                    }
                    add_modifier = {
                        modifier = hsr_credits_planet_modifiers
                        years = -1
                    }
                }
            }
            # 固定年度经营性口径：殖民地税收加原国家财政收入修正；不读余额差或投资市值。
            change_variable = {
                which = hsr_credits_country_year_income_gross
                value = modifier:country_credit_gross_other_mult
            }
            change_variable = { which = hsr_credits_country_trade_storage value = value:mod_year_credits_add }
            set_variable = { which = hsr_finance_last_income value = hsr_credits_country_year_income_gross }
            if = {
                limit = {
                    check_variable = { which = hsr_finance_last_income value < 0 }
                }
                set_variable = { which = hsr_finance_last_income value = 0 }
            }
            set_variable = { which = hsr_finance_annual_expense value = modifier:country_credit_expenses_other_mult }
            change_variable = { which = hsr_finance_annual_expense value = hsr_finance_last_interest_paid }
            hsr_finance_refresh = yes
            remove_modifier = hsr_credits_country_modifiers
            if = {
                limit = {
                    check_variable = { which = hsr_credits_country_trade_storage value >= 1000000 }
                }
                add_modifier = {
                    modifier = hsr_credits_country_modifiers
                    years = -1
                    mult = 10
                }
            }
            else = {
                add_modifier = {
                    modifier = hsr_credits_country_modifiers
                    years = -1
                    mult = value:hsr_add_mult_value|BASEVAR|0.00001|VAR1|hsr_credits_country_trade_storage|
                }
            }
        }
        # HSR_FINANCE_YEAR_END
    }
    trigger = {
        # HSR_SYSTEM_SWITCH_BEGIN
        NOT = {
            has_global_flag = hsr_credit_system_disabled
        }
        # HSR_SYSTEM_SWITCH_END
    }
}
