country_event = {
    id = hsr_eco.100
    is_triggered_only = yes
    hide_window = yes
    trigger = {
        #has_country_flag = hsr_credit_open
    }
    immediate = {
        every_owned_colony = {
            limit = { hsr_credits_is_live_colony = yes }
            #地方政策
            if = {
                limit = {
                    not = {
                        has_deposit = d_colony_credits_slot
                    }
                }
                add_deposit = d_colony_credits_slot
            }
            if = {
                limit = {
                    or = {
                        check_variable = { which = hsr_planet_social value < 1 }
                        check_variable = { which = hsr_planet_social value > 3 }
                    }
                }
                set_variable = { which = hsr_planet_social value = 2 }
            }
            if = {
                limit = {
                    or = {
                        check_variable = { which = hsr_planet_governance value < 1 }
                        check_variable = { which = hsr_planet_governance value > 3 }
                    }
                }
                set_variable = { which = hsr_planet_governance value = 2 }
            }
            if = {
                limit = {
                    or = {
                        check_variable = { which = hsr_planet_military value < 1 }
                        check_variable = { which = hsr_planet_military value > 3 }
                    }
                }
                set_variable = { which = hsr_planet_military value = 2 }
            }
            if = {
                limit = {
                    or = {
                        check_variable = { which = hsr_planet_industrial value < 1 }
                        check_variable = { which = hsr_planet_industrial value > 3 }
                    }
                }
                set_variable = { which = hsr_planet_industrial value = 2 }
            }
            #读取数据
            hsr_credits_read_production = yes
            #计算数据
            if = {
                limit = {
                    check_variable = {
                        which = hsr_credits_net
                        value > 0
                    }
                }
                if = {
                    limit = {
                        root = {has_policy_flag = hsr_basics_tax_revenue_captial}
                    }
                    set_variable = { which = hsr_credits_planet_year_value value = value:hsr_add_mult_value|BASEVAR|0.3|VAR1|hsr_credits_net|  }
                    set_variable = { which = hsr_credits_country_year_value value = value:hsr_add_mult_value|BASEVAR|0.7|VAR1|hsr_credits_net|  }
                }
                else_if = {
                    limit = {
                        root = {has_policy_flag = hsr_basics_tax_revenue_local}
                    }
                    set_variable = { which = hsr_credits_planet_year_value value = value:hsr_add_mult_value|BASEVAR|0.7|VAR1|hsr_credits_net|  }
                    set_variable = { which = hsr_credits_country_year_value value = value:hsr_add_mult_value|BASEVAR|0.3|VAR1|hsr_credits_net|  }
                }
                else_if = {
                    limit = {
                        root = {has_policy_flag = hsr_basics_tax_revenue_average}
                    }
                    set_variable = { which = hsr_credits_planet_year_value value = value:hsr_add_mult_value|BASEVAR|0.5|VAR1|hsr_credits_net|  }
                    set_variable = { which = hsr_credits_country_year_value value = value:hsr_add_mult_value|BASEVAR|0.5|VAR1|hsr_credits_net|  }
                }
                change_variable = { which = hsr_credits_planet_trade_storage value = hsr_credits_planet_year_value }
                prev = {
                    change_variable = { which = hsr_credits_country_trade_storage value = prev.hsr_credits_country_year_value }#总存储
                    #change_variable = { which = hsr_credits_country_year_income_gross value = prev.hsr_credits_country_trade }#年收入
                }
            }
            else_if = {
                limit = {
                    check_variable = {
                        which = hsr_credits_net
                        value < 0
                    }
                }
                change_variable = { which = hsr_credits_planet_trade_storage value = hsr_credits_net }
            }
            #建议显示
            if = {
                limit = {
                    not = {
                        has_modifier = hsr_credits_planet_modifiers
                    }
                }
                add_modifier = {
                    modifier = hsr_credits_planet_modifiers
                    years = -1
                }
            }
        }
        remove_modifier = hsr_credits_country_modifiers
        add_modifier = {
            modifier = hsr_credits_country_modifiers
            years = -1
            mult = value:hsr_add_mult_value|BASEVAR|0.00001|VAR1|hsr_credits_country_trade_storage|
        }
    }
}
