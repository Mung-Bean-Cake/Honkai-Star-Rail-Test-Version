		give_technology = { tech = tech_robomodding_points_2 message = no }
		give_technology = { tech = tech_positronic_implants message = no }
		give_technology = { tech = tech_habitat_1 message = no }

		give_technology = { tech = tech_honkai_star_rail_weapon_t1 message = no }
		give_technology = { tech = tech_honkai_star_rail_hull_t1 message = no }
		give_technology = { tech = tech_honkai_star_rail_power_t1 message = no }
		give_technology = { tech = tech_science_arkship message = no }

		give_technology = { tech = tech_starbase_2 message = no }
		give_technology = { tech = tech_solar_panel_network message = no }
		give_technology = { tech = tech_space_defense_station_1 message = no }
		add_research_option = tech_galactic_archivism
		# set_policy = {	}
		# 像黑塔空间站一样：普通帝国开局在母星(Pegana)上主动生成列车方舟舰队。
		# 不再依赖游牧开局自动给的方舟船，也不再需要 set_country_as_nomad_effect。
		capital_scope = {
			save_event_target_as = astral_start_planet  # 先记住母星，首都迁走后仍指向它
			create_fleet = {
				name = "Astral_Express"
				effect = {
					set_owner = root
					create_ship = {
						name = "Astral_Express" # 关键
						design = "train_planet_fleet_1" # 关键
						effect = {
							starbase = {
								set_starbase_building = {
									slot = 1
									building = arkship_hsr_base_hub
								}
								set_starbase_building = {
									slot = 2
									building = arkship_hsr_advanced_hub
								}
								set_starbase_building = {
									slot = 3
									building = arkship_hsr_unity_hub
								}
							}
							fleet = {
							}
							colony = {
								add_building = building_capital
								while = {
									count = 2
									create_pop_group = {
										size = 100
										species = owner_main_species
									}
								}
								set_capital = yes  # 列车方舟成为首都（母星随后被清殖民）
							}
						}
					}
					set_fleet_stance = evasive
					set_fleet_flag = train_fleet
					set_location = {
						target = prev
						distance = 80
						angle = random

		event_target:astral_start_planet = {
			destroy_colony = yes
			change_pc = pc_volcanic
		}
	}

					NOT = { has_origin = origin_Astral_Express }
					NOT = { has_country_flag = DOT_SR_ae_met }
				}
			}
			from = { dot_sr_ship_start_clock = yes }
