                    hsr_corp_ai_manage_slot = { SLOT = 2 }
                    hsr_corp_ai_manage_slot = { SLOT = 3 }
                }
            }
        }
        # AI一年最多一笔少量市场投资，采用和玩家相同的支付及冷却校验。

            limit = {
                is_ai = yes

                limit = {
                    is_country_type = default

                    NOT = {
                        is_at_war_with = event_target:hsr_corp_actor

                }
                random_owned_colony = {
                    carrier = {
                        set_variable = { which = hsr_corp_order_qty value = 1 }
                        hsr_corp_buy = { SLOT = 1 }
                    }
                }
            }
        }
    }
