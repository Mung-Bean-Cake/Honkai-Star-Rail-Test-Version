country_event = {
    id = DOT_SR_Hunter.2
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        has_origin = origin_StellaronHunter
    }

    immediate = {
        if = {
            limit = {
                has_origin = origin_StellaronHunter
                NOT = { has_technology = tech_shuhu }
            }
            add_research_option = tech_shuhu
            limit = {
                has_origin = origin_StellaronHunter
                NOT = { has_technology = tech_hacker_skill }
            }
            add_research_option = tech_hacker_skill
        }
        random_planet_within_border = {
            limit = { has_planet_flag = hunter }
            remove_building = building_luxury_residence
            remove_building = building_precinct_house
            add_building = building_hunter_ren
            add_building = building_gelamo_military_camp
        }
        begin_event_chain = {
            event_chain = "hunter_discover_situation_chain"
            target = ROOT
        }
        capital_scope = {
            enable_special_project = {
                name = "hunter_discover_situation_1"
                location = this
                owner = root
            }
            enable_special_project = {
                name = "hunter_discover_situation_2"
                location = this
                owner = root
            }
            enable_special_project = {
                name = "hunter_discover_situation_3"
                location = this
                owner = root
            }
        }
    }
}
