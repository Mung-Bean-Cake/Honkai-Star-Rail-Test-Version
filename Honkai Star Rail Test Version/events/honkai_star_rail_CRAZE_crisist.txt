                ethic = ethic_gestalt_consciousness
            }
            type = "Hostile_mid_crisis_country"
            name_list = "DOT_SR_HUMAN1"
            flag = random

			effect = {
				save_global_event_target_as = NAME_tz_menp_1_test_2
			}

				owner = {
					is_ai = no

			}
			random_system = {
				limit = {
					distance = {
					    source = event_target:erxiangleyuan_system
					    min_jumps = 4

				}
				}
				while = {
					count= 3
				create_fleet = {
					effect = {
						set_fleet_flag = NAME_tz_menp_1_test_2
						set_owner = event_target:NAME_tz_menp_1_test_2

							count = 20
							create_ship = {
								name = random
								design = tantao_peitai_ship_design
								graphical_culture = humanoid_01
							}
						}
						fleet_event = { id = craze.11 }
						set_location = {
							target = prev
							distance = 45
							angle = random
						}
					}				
				}
				}
				
				
			}			
		}
	}

	option = {
		name = craze.10_op_1
		set_variable = { 
			which = tz_menp_1_test_2_defeated_count 
			value = 0 
		}
	}
}

fleet_event = {
    id = craze.11
    hide_window = yes
    is_triggered_only = yes

    immediate = {
        every_country = {
            limit = {
                is_ai = no
                is_country_type = default
				has_origin = origin_erxiangleyuan
            }
            save_global_event_target_as = erxiangleyuan_target_country
        }

		event_target:NAME_tz_menp_1_test_2 = {
			set_hostile = event_target:erxiangleyuan_target_country
		}
        
        if = {
            limit = {
                event_target:erxiangleyuan_target_country = {
                    any_owned_fleet = { is_ship_class = shipclass_military }
                }
            }
            event_target:erxiangleyuan_target_country = {
                random_owned_fleet = {
                    limit = { is_ship_class = shipclass_military }
                    save_event_target_as = erxiangleyuan_fleet_target
                }
            }
            auto_follow_fleet = {
                target = event_target:erxiangleyuan_fleet_target
                attack_fleet = yes
            }
        }

        else = {
            event_target:erxiangleyuan_target_country = {
                capital_scope = {
                    save_event_target_as = erxiangleyuan_capital_planet
                }
            }
            auto_move_to_planet = {
                target = event_target:erxiangleyuan_capital_planet
				clear_auto_move_on_arrival = yes	
            }
        }
    }
}
