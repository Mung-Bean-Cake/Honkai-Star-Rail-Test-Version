country_event = {
	id = craze.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		
		check_variable = {
			which = the_craze_variable
			value >=10
		}
	}

	immediate = {
		set_country_flag = the_craze_crisis_1
		random_owned_planet = {
			limit = {is_capital = yes}
			add_modifier = {
				modifier = planet_tantao_pollution_1
				days = -1
			}
		}
		country_event = {
			id = craze.2
		}
	}
}

country_event = {
	id = craze.4
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		#已有殖民星球，否则殖民下方星球
		random_system = {
			limit = {
				owner = {
					is_ai = no
					has_origin = origin_erxiangleyuan
				}
			    has_planet_class = pc_erxiangleyuan
			}
			spawn_system = {
			    initializer = DOT_SR_system_ss1
			        max_distance = 130
			        min_distance = 70
			        hyperlane = yes
			        direction = rimwards
		    }
		}
		

		random_owned_planet = {
			limit= {
				is_capital = no
				distance = {
					source = event_target:erxiangleyuan_sysytem
					min_jumps = 1
					max_jumps = 15
				}
			}
			hidden_effect = {
				save_global_event_target_as = craze_planet_0
				set_planet_flag = craze_planet_0
			}
		}	
	}
}
