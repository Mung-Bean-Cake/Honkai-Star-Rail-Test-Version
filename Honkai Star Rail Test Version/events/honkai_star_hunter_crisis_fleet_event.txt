namespace = crisis_fleet_action
fleet_event = {
    id = crisis_fleet_action.1
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        exists = this
        has_global_flag = fengrao_crisis_started
        owner = {
            is_same_value = event_target:crisis_country
        }
        NOT = {
            is_ship_class = shipclass_transport
        }
    }
    immediate = {
        if = {
            limit = {
                NOT = {
                    any_combatant_fleet = {
                        always = yes
                    }
                }
            }
            crisis_fleet_action = yes
            crisis_fleet_move_toward_colony = yes
        }
        fleet_event = {
            id = crisis_fleet_action.1 days = 15 random = 10
        }
    }
}
fleet_event = {
    id = crisis_fleet_action.2
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        exists = this
        has_global_flag = fengrao_crisis_started
        owner = {
            is_same_value = event_target:crisis_country
        }
        is_ship_class = shipclass_transport
    }
    immediate = {
        if = {
            limit = {
                NOT = {
                    any_combatant_fleet = {
                        always = yes
                    }
                }
            }
            if = {
                limit = {
                    solar_system = {
                        any_system_planet = {
                            exists = controller
                            NOT = { controller = { is_same_value = event_target:crisis_country } }
                            owner = { is_hostile = event_target:crisis_country }
                            OR = {
                                is_colony = yes
                                is_under_colonization = yes
                            }
                        }
                    }
                }
                fleet_event = {
                    id = crisis_fleet_action.3 days = 2
                }
            }
            else = {
                crisis_fleet_transport_action = yes
                crisis_fleet_move_toward_colony = yes
            }
        }
        fleet_event = {
            id = crisis_fleet_action.2 days = 15 random = 10
        }
    }
}
fleet_event = {
    id = crisis_fleet_action.3
    hide_window = yes
    is_triggered_only = yes
    trigger = {
        exists = this
        has_global_flag = fengrao_crisis_started
        owner = { is_same_value = event_target:crisis_country }
        is_ship_class = shipclass_transport
    }
    immediate = {
        solar_system = {
            random_system_planet = {
                limit = {
                    exists = controller
                    NOT = { controller = { is_same_value = event_target:crisis_country } }
                    owner = { is_hostile = event_target:crisis_country }
                    OR = {
                        is_colony = yes
                        is_under_colonization = yes
                    }
                }
                save_event_target_as = crisis_land_target
            }
        }
        if = {
            limit = { exists = event_target:crisis_land_target }
            queue_actions = {
                find_closest_planet = {
                    trigger = {
                        is_same_value = event_target:crisis_land_target
                    }
                    found_planet = {
                        move_to = this
                        orbit_planet = this
                        land_armies = this
                    }
                }
                
            }
        }
    }
}
