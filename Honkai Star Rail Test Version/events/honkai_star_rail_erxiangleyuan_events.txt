#被动效果
country_event = {
	id = xiaoqiaosi.3000
	title = xiaoqiaosi.3000_title
	desc = xiaoqiaosi.3000_desc
	is_triggered_only = yes
	#hide_window = yes

	trigger = {
		has_country_flag = is_huanyu_time
	}

	immediate = {
		#leader_huanyunv_t4
		if = {
			limit = {any_owned_leader = {has_leader_flag = flag_leader_huanyunv_t4}}
			add_modifier = {
				modifier = huanyu_leader_huanyunv_t4
				days = 180
			}
		}
		#leader_huanyunv_t5
		if = {
			limit = {any_owned_leader = {has_leader_flag = flag_leader_huanyunv_t5}}
			add_modifier = {
				modifier = huanyu_leader_huanyunv_t5
				days = 90
			}
		}
		#leader_huanyunv_t6
		if = {
			limit = {any_owned_leader = {has_leader_flag = flag_leader_huanyunv_t6} has_country_flag = is_huanyu_time}
			change_variable = {
				which = num_leader_huanyunv_t6
				value = 1
			}
		}
		#leader_huohua_t2
		if = {
			limit = {any_owned_leader = {has_leader_flag = huohua} has_country_flag = is_huanyu_time}
			change_variable = {
				which = num_leader_huohua_t2
				value = 1
			}
		}
		#leader_huohua_t5
		if = {
			limit = {any_owned_leader = {has_leader_flag = flag_leader_huohua_t5} has_country_flag = is_huanyu_time}
			change_variable = {
				which = xiaodian
				value = 10
			}
		}
		#leader_feiying_t2
		if = {
			limit = {any_owned_leader = {has_leader_flag = feiying} has_country_flag = is_show_time_feiying}
			change_variable = {
				which = num_leader_feiying_t2
				value = 1
			}
		}
		#leader_feiying_t4
		if = {
			limit = {any_owned_leader = {has_leader_flag = feiying} has_country_flag = is_huanyu_time}
			add_modifier = {
				modifier = huanyu_leader_feiying_t4
				days = 180
			}
		}
		#leader_langzun_t1
		#回复笑点
		if = {
			limit = {any_owned_leader = {has_leader_flag = langzun}}
			change_variable = {
				which = xiaodian
				value = this.owner.num_leader_langzun_t1
			}
			change_variable = {
				which = num_leader_langzun_t2
				value = this.owner.num_leader_langzun_t1
			}
		}
		#被动叠层
		if = {
			limit = {any_owned_leader = {has_leader_flag = langzun} has_country_flag = is_huanyu_time}
			change_variable = {
				which = num_leader_langzun_t1
				value = 1
			}
		}	
		
	}
	after = {
		remove_country_flag = is_huanyu_time
	}
}
