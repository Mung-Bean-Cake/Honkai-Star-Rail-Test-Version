Star_Rail_arkship_grand_archive = {
	icon = "GFX_arkship_grand_archive"
	section = "Star_Rail_ARKSHIP_LIVING_SPACE_MODULE"
	construction_days = 1800
	starbase_type = arkship
	category = sm_utility_cat

	empire_limit = { base = 1 }

	potential = {
		Star_Rail_arkship_moudles_and_buildings_potential_built = yes
	}
	cannot_fit_on_slots = { spinal_mount_1 spinal_mount_2 }

	resources = {
		category = starbase_arkship_upgrades
		cost = { alloys = @arkship_cost_alloys_500 volatile_motes = @arkship_cost_strategic_100 }
		upkeep = { energy = @arkship_upkeep_energy_10 }
	}

	country_modifier = { country_vivarium_max_capacity_add = @grand_archive_vivarium_capacity }

	on_finished = {
		owner = {
			if = { limit = { has_country_flag = grand_archive_destroyed } remove_country_flag = grand_archive_destroyed }
			set_country_flag = has_Star_Rail_arkship_grand_archive_module
			country_event = { id = grand_archive.4120 }
			country_event = { id = grand_archive.4199 }
			if = { limit = { has_origin = origin_treasure_hunters } country_event = { id = grand_archive.8090 scopes = { from = prev.solar_system } } }
		}
	}
	on_destroyed = {
		owner = {
			remove_country_flag = ga_destroyed_contains_space_fauna
			if = { limit = { num_vivarium_slots > 0 } set_country_flag = ga_destroyed_contains_space_fauna release_all_vivarium_fauna = { LOCATION = prev.solar_system } }
			set_country_flag = grand_archive_destroyed
			remove_country_flag = has_Star_Rail_arkship_grand_archive_module
		}
	}

	ai_weight = { weight = 100 }
}
