ap_xunlie_hunt = {
	potential = {
		OR = {
			has_origin = origin_StellaronHunter
			has_origin = origin_Xianzhou
		}
	}
	possible = {
		has_tradition = tr_xunlie_hunt_5
		custom_tooltip = {
			fail_text = "requires_independence"
			is_subject = no
		}
		if = {	
			OR = {
				owner = { has_origin = origin_StellaronHunter}	
			}	
			custom_tooltip = {
				fail_text = "requires_ascension_perks_2"
				num_ascension_perks >= 2
			}
			NOR = {
				has_civic = civic_machine_servitor
				has_valid_civic = civic_hive_familiar_face
			}
		}
		NOT = {
			has_ascension_perk = ap_wings_of_stars
		}
		custom_tooltip = {
			fail_text = "not_wings_of_stars"
			is_subject = no
		}
	}
	on_enabled = {
		add_research_option = tech_xz_thunder_crossbow_titan
	}
	modifier = {
		country_naval_cap_add = 2000
		country_war_exhaustion_mult = -0.1
		diplo_action_acceptance_add = 300
	}
	ai_weight= {
		factor = 0
		modifier = {
			has_origin = origin_Xianzhou
			factor = 100
		}
	}
}
