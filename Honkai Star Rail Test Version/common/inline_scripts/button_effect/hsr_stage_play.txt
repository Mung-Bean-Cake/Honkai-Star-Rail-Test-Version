    allow = {
        custom_tooltip = {
            fail_text = need_selected
            hidden_trigger = {
                from = {
                    has_country_flag = $civic_name$
                }
            }
        }
        custom_tooltip = {
            fail_text = need_selected_slot
            hidden_trigger = {
                from = {
                    is_path_slot = yes
                }
            }
        }
    }
    potential = {
        always = yes
    }
    effect = {
        custom_tooltip = reset_civic
        hidden_effect = {
            from = {
                hsr_civic_reset = {civic_name = $civic_name$ value = $value$}
            }
        }
    }
}
