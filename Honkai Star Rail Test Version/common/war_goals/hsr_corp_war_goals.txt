# 状态：0空、1建设、2活动、3危机、4清算。未经完整游戏验收。
# 这些目标只在新宣战时选择；没有任何事件替换既有战争目标。
wg_hsr_corp_currency_war = {
    casus_belli = cb_hsr_corp_currency_war
    hide = no_cb
    surrender_acceptance = -100
    potential = {
        has_origin = origin_hsr_ipc
        has_ascension_perk = ap_hsr_corp_currency_war
    }
    possible = {
        has_origin = origin_hsr_ipc
        has_ascension_perk = ap_hsr_corp_currency_war
        from = {
            is_country_type = default
            is_subject = no
            is_galactic_emperor = no
            is_galactic_custodian = no
        }
    }
    on_accept = {
        hidden_effect = {
            hsr_corp_war_settlement = yes
            from = {
                if = {
                    limit = {
                        is_subject = no
                        is_galactic_emperor = no
                        is_galactic_custodian = no
                    }
                    set_subject_of = {
                        who = root
                        preset = preset_ipc_arc_subject
                    }
                }
            }
        }
    }
    on_status_quo = {
        custom_tooltip = hsr_corp_no_asset_transfer
    }
    ai_weight = {
        weight = 20
    }
}

wg_hsr_corp_protect_assets = {
    casus_belli = cb_hsr_corp_protect_assets
    hide = no_cb
    surrender_acceptance = -25
    potential = {
        has_country_flag = hsr_corp_loss@from
    }
    possible = {
        has_country_flag = hsr_corp_loss@from
    }
    on_accept = {
        hidden_effect = {
            if = {
                limit = {
                    has_country_flag = hsr_corp_loss@from
                    check_variable = { which = hsr_corp_damage_claim value > 0 }
                    exists = event_target:hsr_corp_offender@this
                    event_target:hsr_corp_offender@this = {
                        is_same_value = from
                    }
                }
                save_event_target_as = hsr_corp_victor
                from = {
                    save_event_target_as = hsr_corp_defeated
                }
                hsr_corp_compensate = { LIMIT = hsr_corp_damage_claim }
                remove_country_flag = hsr_corp_loss@from
                set_variable = { which = hsr_corp_damage_claim value = 0 }
                clear_global_event_target = hsr_corp_offender@this
            }
        }
    }
    on_status_quo = {
        custom_tooltip = hsr_corp_no_asset_transfer
    }
    ai_weight = {
        weight = 10
    }
}
