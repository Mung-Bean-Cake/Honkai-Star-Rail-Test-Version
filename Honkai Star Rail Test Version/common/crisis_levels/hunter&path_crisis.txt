wings_of_stars_level_1 = {
	required_crisis_currency = 0
	perks = {
		wings_of_stars_level_1_1
		wings_of_stars_level_1_2
		wings_of_stars_level_1_3
		wings_of_stars_level_1_4
	}
	on_unlock = {
		hidden_effect = {
			owner = {
				set_country_flag = wings_of_stars_level_1
				country_event = {id = path_country.41}
			}
			if = {
				limit = {
					exists = event_target:origin_Xianzhou
				}
				add_opinion_modifier = {
					modifier = wings_relation_boost
					who = event_target:origin_Xianzhou
				}
				event_target:origin_Xianzhou = {
					add_opinion_modifier = {
						modifier = wings_relation_boost
						who = prev
					}
				}
			}
		}
	}
}
