hunter_fleet_contribution_medium = {
	allow = {
		custom_tooltip = {
			NOT = {
				AND = {
					NOT = { has_federation_law = hunter_fleet_contribution_medium }
					has_modifier = hunter_federation_law_category_fleet_contribution_cooldown
					has_any_federation_law_in_category = law_hunter_fleet_contribution
				}
			}
		}
	}
	potential = {
		OR = {
			has_federation_type = xunlie_gov_federation
			has_federation_type = new_order_federation
		}
	}
	required_centralization = {
		base = 0
	}
	modifier =	{
		description = "hunter_fleet_contribution_medium_effect"
		federation_fleet_contribution_factor = 0.4
		federation_fleet_cap_add = 5000
	}
	cohesion_growth = -0.5
	on_enact = {
		if = {
			limit = { has_federation_flag = enable_federation_cooldowns }
			add_modifier = {
				modifier = hunter_federation_law_category_fleet_contribution_cooldown
				days = @cooldown
			}
		}
		hidden_effect = {
			if = {
				limit = {
					has_federation_flag = no_federation_fleet
				}
				remove_federation_flag = no_federation_fleet
				set_timed_federation_flag = {
					flag = no_federation_fleet_build_right_cooldown
					days = 10
				}
			}
		}
		every_member = {
			if = {
				limit = {
					has_modifier = federation_naval_cap_contribution_mult_large
				}
				remove_modifier = federation_naval_cap_contribution_mult_large
			}
			add_modifier = {
				modifier = federation_naval_cap_contribution_mult_medium
				years = 100
			}
		}
	}
	ai_weight = {
		base = 40
	}
}
