# 仅在首次打开新载体时初始化三槽；已初始化的当前公司绝不重置。
hsr_corp_initialize = {
    optimize_memory
    if = {
        limit = { NOT = { is_variable_set = hsr_credits_planet_trade_storage } }
        set_variable = { which = hsr_credits_planet_trade_storage value = 0 }
    }
    if = {
        limit = {
            NOT = {
                has_carrier_flag = hsr_corp_data_initialized
            }
        }
        hsr_corp_initialize_slot = { SLOT = 1 }
        hsr_corp_initialize_slot = { SLOT = 2 }
        hsr_corp_initialize_slot = { SLOT = 3 }
        set_carrier_flag = hsr_corp_data_initialized
    }
}

# 持久引用使用原版支持的动态全局事件目标；局部目标仅用于一次事务。
hsr_corp_resolve_parties = {
    optimize_memory
    save_event_target_as = hsr_corp_hq
    owner = {
        save_event_target_as = hsr_corp_founder
        save_event_target_as = hsr_corp_foreign
        save_event_target_as = hsr_corp_controller
    }
    if = {
        limit = {
            exists = event_target:hsr_corp_founder_s$SLOT$@this
        }
        event_target:hsr_corp_founder_s$SLOT$@this = {
            save_event_target_as = hsr_corp_founder
            save_event_target_as = hsr_corp_controller
        }
    }
    if = {
        limit = {
            exists = event_target:hsr_corp_foreign_s$SLOT$@this
        }
        event_target:hsr_corp_foreign_s$SLOT$@this = {
            save_event_target_as = hsr_corp_foreign
        }
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_s$SLOT$_foreign_control value = 1 }
            exists = event_target:hsr_corp_foreign_s$SLOT$@this
        }
        event_target:hsr_corp_foreign_s$SLOT$@this = {
            save_event_target_as = hsr_corp_controller
        }
    }
    if = {
        limit = {
            has_carrier_flag = hsr_corp_rights_suspended_s$SLOT$
        }
        owner = {
            save_event_target_as = hsr_corp_controller
        }
    }
}

# 单次创建事务。支付与快照在同一个有资格/资金保护的块内完成；不自动混合账户。
hsr_corp_create = {
    optimize_memory
    set_variable = { which = hsr_corp_creation_success value = 0 }
    hsr_corp_initialize = yes
    hsr_corp_quote_creation = yes
    if = {
        limit = {
            hsr_corp_creation_allowed = { SLOT = $SLOT$ }
        }
        set_variable = { which = hsr_corp_creation_success value = 1 }
        remove_carrier_flag = hsr_corp_transfer_in_progress
        hsr_corp_initialize_slot = { SLOT = $SLOT$ }
        set_variable = { which = hsr_corp_s$SLOT$_control_notice_initialized value = 1 }
        save_event_target_as = hsr_corp_hq
        every_country = {
            every_war = {
                remove_war_flag = hsr_corp_aid_s$SLOT$@event_target:hsr_corp_hq
            }
            remove_country_flag = hsr_corp_crisis_aid_s$SLOT$@event_target:hsr_corp_hq
        }
        event_target:global_event_country = {
            if = {
                limit = {
                    NOT = {
                        is_variable_set = hsr_corp_next_id
                    }
                }
                set_variable = { which = hsr_corp_next_id value = 0 }
            }
            change_variable = { which = hsr_corp_next_id value = 1 }
        }
        set_variable = { which = hsr_corp_s$SLOT$_id value = event_target:global_event_country.hsr_corp_next_id }
        set_variable = { which = hsr_corp_s$SLOT$_state value = 1 }
        set_variable = { which = hsr_corp_s$SLOT$_level value = hsr_corp_setup_level }
        set_variable = { which = hsr_corp_s$SLOT$_primary value = hsr_corp_setup_primary }
        set_variable = { which = hsr_corp_s$SLOT$_regime value = hsr_corp_setup_regime }
        set_variable = { which = hsr_corp_s$SLOT$_loyalty value = 50 }
        set_variable = { which = hsr_corp_s$SLOT$_founder_shares value = hsr_corp_quote_share }
        set_variable = { which = hsr_corp_s$SLOT$_pool_shares value = 100 }
        subtract_variable = { which = hsr_corp_s$SLOT$_pool_shares value = hsr_corp_s$SLOT$_founder_shares }
        set_variable = { which = hsr_corp_s$SLOT$_cash value = hsr_corp_quote }
        multiply_variable = { which = hsr_corp_s$SLOT$_cash value = value:hsr_corp_initial_cash }
        set_variable = { which = hsr_corp_s$SLOT$_market_reserve value = hsr_corp_quote }
        multiply_variable = { which = hsr_corp_s$SLOT$_market_reserve value = value:hsr_corp_initial_market }
        set_variable = { which = hsr_corp_s$SLOT$_share_price value = hsr_corp_quote }
        divide_variable = { which = hsr_corp_s$SLOT$_share_price value = 100 }
        set_variable = { which = hsr_corp_s$SLOT$_founder_basis value = hsr_corp_s$SLOT$_share_price }
        owner = {
            save_global_event_target_as = hsr_corp_founder_s$SLOT$@event_target:hsr_corp_hq
            save_global_event_target_as = hsr_corp_notified_controller_s$SLOT$@event_target:hsr_corp_hq
            save_global_event_target_as = hsr_corp_payer_s$SLOT$@event_target:hsr_corp_hq
        }
        if = {
            limit = { exists = event_target:hsr_corp_foreign_s$SLOT$@this }
            clear_global_event_target = hsr_corp_foreign_s$SLOT$@this
        }
        if = {
            limit = {
                check_variable = { which = hsr_corp_setup_payment value = 0 }
            }
            set_variable = { which = hsr_corp_s$SLOT$_paid_trade value = hsr_corp_quote }
            set_variable = { which = hsr_corp_charge value = hsr_corp_quote }
            multiply_variable = { which = hsr_corp_charge value = -1 }
            owner = {
                add_resource = { trade = 1 mult = prev.hsr_corp_charge }
            }
        }
        else = {
            set_variable = { which = hsr_corp_s$SLOT$_paid_national value = hsr_corp_quote_national }
            set_variable = { which = hsr_corp_s$SLOT$_paid_local value = hsr_corp_quote_local }
            subtract_variable = { which = hsr_credits_planet_trade_storage value = hsr_corp_quote_local }
            owner = {
                subtract_variable = { which = hsr_credits_country_trade_storage value = prev.hsr_corp_quote_national }
            }
        }
        if = {
            limit = {
                check_variable = { which = hsr_corp_setup_regime value >= 3 }
            }
            set_variable = { which = hsr_corp_s$SLOT$_listed value = 1 }
        }
        if = {
            limit = {
                check_variable = { which = hsr_corp_setup_level value = 4 }
            }
            set_variable = { which = hsr_corp_s$SLOT$_origin_bonus value = 0.1 }
        }

                check_variable = { which = hsr_corp_setup_level value = 7 }

        }
        set_carrier_flag = hsr_corp_has_records
        set_timed_carrier_flag = { flag = hsr_corp_building_s$SLOT$ days = hsr_corp_quote_days }
        carrier_event = {
            id = hsr_corp.21$SLOT$
            days = hsr_corp_quote_days
        }
        hsr_corp_refresh_jobs = yes
    }
}

# 页内确认与独立事件共用事务；成功返回地方列表，失败保留表单，不在此新开窗口。
hsr_corp_confirm_creation = {
    optimize_memory
    set_variable = { which = hsr_corp_creation_success value = 0 }
    if = {
        limit = {
            check_variable = { which = hsr_corp_page value = 3 }
            check_variable = { which = hsr_corp_setup_slot value = 1 }
        }
        hsr_corp_create = { SLOT = 1 }
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_page value = 3 }
            check_variable = { which = hsr_corp_setup_slot value = 2 }
        }
        hsr_corp_create = { SLOT = 2 }
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_page value = 3 }
            check_variable = { which = hsr_corp_setup_slot value = 3 }
        }
        hsr_corp_create = { SLOT = 3 }
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_creation_success value = 1 }
        }
        set_variable = { which = hsr_corp_page value = 1 }
    }
}

# 定时事件和月度修复共用完成入口；状态与到期标记使重复/旧延迟事件无效。
hsr_corp_finish_building = {
    optimize_memory
    if = {
        limit = {
            # HSR_SYSTEM_SWITCH_BEGIN
            NOT = {
                has_global_flag = hsr_currency_war_disabled
            }
            # HSR_SYSTEM_SWITCH_END
            hsr_corp_valid_hq = yes
            check_variable = { which = hsr_corp_s$SLOT$_state value = 1 }
            NOT = {
                has_carrier_flag = hsr_corp_building_s$SLOT$
            }
        }
        if = {
            limit = {
                exists = event_target:hsr_corp_payer_s$SLOT$@this
                OR = {
                    NOT = {
                        check_variable = { which = hsr_corp_s$SLOT$_regime value = 2 }
                    }
                    hsr_corp_cooperative_allowed = yes
                }
            }
            set_variable = { which = hsr_corp_s$SLOT$_state value = 2 }
            hsr_corp_notify_lifecycle = { SLOT = $SLOT$ KIND = CREATED }
            hsr_corp_refresh_jobs = yes
        }
        else = {
            hsr_corp_close_slot = { SLOT = $SLOT$ }
        }
    }
}

# 返还原始快照80%；创建国失效时显式退给当前总部监管国，地方部分仍退本地。
hsr_corp_refund_building = {
    optimize_memory
    hsr_corp_resolve_parties = { SLOT = $SLOT$ }
    owner = {
        save_event_target_as = hsr_corp_refund_country
    }
    if = {
        limit = {
            exists = event_target:hsr_corp_payer_s$SLOT$@this
        }
        event_target:hsr_corp_payer_s$SLOT$@this = {
            save_event_target_as = hsr_corp_refund_country
        }
    }
    set_variable = { which = hsr_corp_refund value = hsr_corp_s$SLOT$_paid_trade }
    multiply_variable = { which = hsr_corp_refund value = value:hsr_corp_refund }
    event_target:hsr_corp_refund_country = {
        add_resource = { trade = 1 mult = prev.hsr_corp_refund }
    }
    set_variable = { which = hsr_corp_refund value = hsr_corp_s$SLOT$_paid_national }
    multiply_variable = { which = hsr_corp_refund value = value:hsr_corp_refund }
    event_target:hsr_corp_refund_country = {
        change_variable = { which = hsr_credits_country_trade_storage value = prev.hsr_corp_refund }
    }
    set_variable = { which = hsr_corp_refund value = hsr_corp_s$SLOT$_paid_local }
    multiply_variable = { which = hsr_corp_refund value = value:hsr_corp_refund }
    change_variable = { which = hsr_credits_planet_trade_storage value = hsr_corp_refund }
}

# 唯一注销入口：先置清算锁，再一次结算、清槽和重算；不操作任何舰船。
hsr_corp_close_slot = {
    optimize_memory
    if = {
        limit = {
