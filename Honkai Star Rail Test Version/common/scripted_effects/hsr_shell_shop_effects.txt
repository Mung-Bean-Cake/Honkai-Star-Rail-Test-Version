hsr_shell_shop_initialize = {
    hidden_effect = {
        if = {
            limit = {
                NOT = {
                    is_variable_set = yuzhousuipian
                }
            }
            set_variable = { which = yuzhousuipian value = 0 }
        }
        if = {
            limit = {
                NOT = {
                    OR = {
                        check_variable = { which = hsr_shell_shop_resource value = 1 }
                        check_variable = { which = hsr_shell_shop_resource value = 2 }
                        check_variable = { which = hsr_shell_shop_resource value = 3 }
                        check_variable = { which = hsr_shell_shop_resource value = 4 }
                        check_variable = { which = hsr_shell_shop_resource value = 5 }
                        check_variable = { which = hsr_shell_shop_resource value = 6 }
                        check_variable = { which = hsr_shell_shop_resource value = 7 }
                        check_variable = { which = hsr_shell_shop_resource value = 8 }
                        check_variable = { which = hsr_shell_shop_resource value = 9 }
                        check_variable = { which = hsr_shell_shop_resource value = 10 }
                        check_variable = { which = hsr_shell_shop_resource value = 11 }
                    }
                }
            }
            set_variable = { which = hsr_shell_shop_resource value = 1 }
        }
        if = {
            limit = {
                NOT = {
                    OR = {
                        check_variable = { which = hsr_shell_shop_lots value = 1 }
                        check_variable = { which = hsr_shell_shop_lots value = 5 }
                        check_variable = { which = hsr_shell_shop_lots value = 10 }
                    }
                }
            }
            set_variable = { which = hsr_shell_shop_lots value = 1 }
        }
        hsr_shell_shop_refresh = yes
    }
}
