# ------------------------------------------------------------
# 裂隙坍缩为虫洞，帝国获得虫洞科技（幂等）
# 在「发现者帝国」作用域内调用
# ------------------------------------------------------------
DOTGI_open_teyvat_wormhole = {
    optimize_memory
	if = {
		limit = {
			NOT = { has_country_flag = DOTGI_void_signal_resolved }
		}
		hidden_effect = {
			set_country_flag = DOTGI_void_signal_resolved

			# 1) 裂隙坍缩（裂隙所在星系 = DOTGI_teyvat_rift_system，开局生成时记下）
			if = {
				limit = {
					exists = event_target:DOTGI_teyvat_rift_system
					event_target:DOTGI_teyvat_rift_system = { has_astral_rift = yes }
				}
				event_target:DOTGI_teyvat_rift_system = {
					astral_rift = {
						destroy_astral_rift = this
					}
				}
			}

			# 2) 裂隙端与提瓦特端各生成一个天然虫洞
			if = {
				limit = { exists = event_target:DOTGI_teyvat_rift_system }
				event_target:DOTGI_teyvat_rift_system = {
					if = {
						limit = { NOT = { has_natural_wormhole = yes } }
						spawn_natural_wormhole = {
							bypass_type = wormhole
							random_pos = yes
							orbit_angle = 360
						}
					}
				}
			}
			if = {
				limit = { exists = event_target:DOTGI_teyvat_system }
				event_target:DOTGI_teyvat_system = {
					if = {
						limit = { NOT = { has_natural_wormhole = yes } }
						spawn_natural_wormhole = {
							bypass_type = wormhole
							random_pos = yes
							orbit_angle = 360
						}
					}
				}
				# 3) 连接两端
				if = {
					limit = { exists = event_target:DOTGI_teyvat_rift_system }
					event_target:DOTGI_teyvat_rift_system = {
						link_wormholes = event_target:DOTGI_teyvat_system
					}
				}
			}

			# 4) 兜底：确保虫洞科技与虫洞认知在手（正常流程在发现裂隙时就已配发）
			give_technology = { tech = tech_wormhole_stabilization message = no }
			add_seen_bypass_type = wormhole
			set_country_flag = DOTGI_wormhole_opened
		}
	}
}

# ------------------------------------------------------------
# 推进【来自虚空的求救信号】局势（裂隙探索事件调用）
# 用法：DOTGI_add_distress_progress = { VALUE = 25 }
# ------------------------------------------------------------
DOTGI_add_distress_progress = {
    optimize_memory
	hidden_effect = {
		if = {
			limit = {
				any_situation = { is_situation_type = DOTGI_distress_signal_situation }
			}
			every_situation = {
				limit = { is_situation_type = DOTGI_distress_signal_situation }
				add_situation_progress = $VALUE$
			}
		}
	}
}

# ------------------------------------------------------------
# 把【来自虚空的求救信号】局势推到完成
# ------------------------------------------------------------
DOTGI_complete_distress_signal = {
    optimize_memory
	if = {
		limit = {
			any_situation = { is_situation_type = DOTGI_distress_signal_situation }
		}
		every_situation = {
			limit = { is_situation_type = DOTGI_distress_signal_situation }
			set_situation_progress = 100
		}
	}
}

# =====================================================================
# 授予提瓦特方舟「死兆星号」
# 作用域：提瓦特国家（DOTGI_teyvat.220 觉醒时调用一次）
# event_target:DOTGI_teyvat_planet = 提瓦特母星
# =====================================================================
DOTGI_grant_alcor_ark = {
	if = {
		limit = { NOT = { has_country_flag = DOTGI_alcor_granted } }
		set_country_flag = DOTGI_alcor_granted
		create_fleet = {
			name = DOTGI_alcor_fleet_name
			effect = {
				set_owner = event_target:DOTGI_teyvat_country
				set_name = DOTGI_alcor_fleet_name
				# 镜像 astral_express.1 的列车方舟生成模式：
				# 方舟舰 create_ship 必须在 effect 里手动初始化 starbase（星基模块/建筑）
				# 与 colony（舰体殖民地的建筑+人口），否则只有一具空壳、没有方舟效果。
				create_ship = {
					name = DOTGI_alcor_ship_name
					design = "Alcor"
					effect = {
						set_graphical_culture = civilian_arkship
						# 星基模块与建筑：直接复用原版游牧方舟初始化效果（船作用域）
						initialize_arkship_starbase_effect = yes
						# 舰体殖民地：首都建筑 + 提瓦特人口（照抄 astral_express.1 的 colony 块）
						colony = {
							add_building = building_capital
							while = {
								count = 2
								create_pop_group = {
									size = 100
									species = owner_main_species
								}
							}
						}
					}
				}
				set_fleet_stance = evasive
				set_fleet_flag = DOTGI_alcor_fleet
				if = {
					limit = { exists = event_target:DOTGI_teyvat_planet }
					set_location = {
						target = event_target:DOTGI_teyvat_planet
						distance = 60
						angle = random
						direction = in_system
					}
				}
				else = {
					set_location = {
						target = event_target:DOTGI_teyvat_country.capital_scope
						distance = 60
						angle = random
						direction = in_system
					}
				}
			}
		}
	}
}
