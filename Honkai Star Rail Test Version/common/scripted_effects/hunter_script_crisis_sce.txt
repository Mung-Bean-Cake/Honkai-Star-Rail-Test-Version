crisis_fleet_action = {
	optimize_memory
	#在当前星系寻找是否存在敌对舰队
	if = {
		limit = {
			root = {
				exists = solar_system
				solar_system = { any_ship_in_system = {
					exists = owner
					owner = { is_hostile = event_target:crisis_country }
					is_disabled = no is_ship_class = shipclass_military
				} }
			}
		}
		root = {
			solar_system = {
				random_fleet_in_system = {
					limit = {
						exists = owner
						owner = { is_hostile = event_target:crisis_country }
						any_owned_ship = { is_disabled = no is_ship_class = shipclass_military }
					}
					prevprev = { auto_follow_fleet = { target = prev attack_fleet = yes } }
				}
			}
		}
	}
	#在当前星系寻找是否存在无己方单位的敌对星球
	else_if = {
		limit = {
			root = {
				exists = solar_system
				solar_system = {
					any_system_planet = {
						exists = controller
						NOT = {
							controller = {
								is_same_value = event_target:crisis_country
							}
						}
						owner = { is_hostile = event_target:crisis_country }
						OR = {
							is_colony = yes
							is_under_colonization = yes
						}
						count_fleet_in_orbit = {
							count = 0
							limit = {
								owner = { is_same_value = event_target:crisis_country }
								is_ship_class = shipclass_military
							}
						}
					}
				}
			}
		}
		root = {
			solar_system = {
				random_system_planet = {
					limit = {
						exists = controller
						NOT = {
							controller = {
								is_same_value = event_target:crisis_country
							}
						}
						owner = { is_hostile = event_target:crisis_country }
						OR = {
							is_colony = yes
							is_under_colonization = yes
						}
						count_fleet_in_orbit = {
							count = 0
							limit = {
								owner = { is_same_value = event_target:crisis_country }
								is_ship_class = shipclass_military
							}
						}
					}
					prevprev = { auto_move_to_planet = { target = prev clear_auto_move_on_arrival = yes } }
				}
			}
		}
	}
	#在当前星系寻找是否存在敌对登陆船
	else_if = {
		limit = {
			root = {
				exists = solar_system
				solar_system = { any_ship_in_system = {
					exists = owner
					owner = { is_hostile = event_target:crisis_country }
					is_disabled = no is_ship_class = shipclass_transport
				} }
			}
		}
		root = {
			solar_system = {
				random_fleet_in_system = {
					limit = {
						exists = owner
						owner = { is_hostile = event_target:crisis_country }
						any_owned_ship = { is_disabled = no is_ship_class = shipclass_transport }
					}
					prevprev = { auto_follow_fleet = { target = prev attack_fleet = yes } }
				}
			}
		}
	}
	#在周围星系寻找是否存在敌对单位（空间站，舰队，殖民地，登陆船）
	else_if = {
		limit = {
			solar_system = { any_neighbor_system = { OR = {
				AND = {
					exists = starbase
					starbase = { exists = owner owner = { is_hostile = event_target:crisis_country } }
				}
				any_ship_in_system = {
					exists = owner
					owner = { is_hostile = event_target:crisis_country }
					is_disabled = no is_ship_class = shipclass_military
				}
