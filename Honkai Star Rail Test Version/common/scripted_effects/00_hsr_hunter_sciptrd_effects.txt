fengrao_shield_burst_small_effect = {
	every_country = {
		every_owned_ship = {
			limit = {
				distance = { 
					source = event_target:fengrao_burst_center
					same_solar_system = yes
					type = euclidean
					max_distance <= $RADIUS|100$
				}
				not = { owner = {
					is_same_value = event_target:fengrao_burst_owner
				} }
			}
			reduce_shield = $SHIELD_DAMAGE|40000$
			reduce_hp = $DAMAGE|20000$
		}
	}
}

create_fengraomin_country = {
	event_target:fengraomin_system = {
		random_system_planet = {
			limit = { $HOME_PLANET_TRIGGER|always$ = yes }
			save_event_target_as = fengraomin_home
			create_country = {
				name = "NAME_fengraomin"
				type = pirate
				species = event_target:owner_species
				name_list = "DOT_SR_fengraomin"
				flag = {
					background = {
						category = "backgrounds"
						file = "00_solid.dds"
					}
					colors ={
						"black"
						"blue"
						"null"
						"null"
					}
				}
			}
			event_target:fengraomin_system = {
				set_name = random
				randomize_flag_symbol = "pirate"
				every_country = {
					establish_communications_no_message = prev
				}
			}
		}
	}
}
create_fengraomin_fleet = {
	event_target:fengraomin_system = {
		create_leader = {
			class = commander
			species = DOT_SR_fengraomin
			name = random
			skill = 1
		}
		create_fleet = {
			name = "NAME_Fengraomin_Fleet"
			effect = {
				set_owner = event_target:fengraomin_system
				if = {
					limit = {
						years_passed < 20
					}
					create_fengraomin_ships_0 = yes
				}
				if = {
					limit = {
						years_passed > 40
					}
					create_pirate_ships_1 = yes
				}
				if = {
					limit = {
						years_passed > 60
					}
					create_pirate_ships_2 = yes
				}
				if = {
					limit = {
						years_passed > 80
					}
					create_pirate_ships_3 = yes
				}
				if = {
					limit = {
						years_passed > 100
					}
					create_pirate_ships_4 = yes
				}
				if = {
					limit = {
						years_passed > 110
					}
					create_pirate_ships_5 = yes
				}
				if = {
					limit = {
						years_passed > 120
					}
					create_pirate_ships_6 = yes
				}
				if = {
					limit = {
						years_passed > 130
					}
					create_pirate_ships_7 = yes
				}
				if = {
					limit = {
						years_passed > 140
					}
					create_pirate_ships_8 = yes
				}
				if = {
					limit = {
						years_passed > 150
					}
					create_pirate_ships_9 = yes
				}
				if = {
					limit = {
						years_passed > 160
					}
					create_pirate_ships_10 = yes
				}
				assign_leader = last_created_leader
				set_location = event_target:fengraomin_home
				set_fleet_stance = aggressive
				set_aggro_range_measure_from = self
				set_aggro_range = 800
			}
		}
	}
}
