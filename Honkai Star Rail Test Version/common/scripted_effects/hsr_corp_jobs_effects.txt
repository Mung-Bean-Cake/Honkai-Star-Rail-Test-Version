    optimize_memory
    set_variable = { which = hsr_corp_occupied value = 0 }

        limit = {
            check_variable = { which = hsr_corp_s1_state value > 0 }

        limit = {
            check_variable = { which = hsr_corp_s2_state value > 0 }

    set_variable = { which = hsr_corp_total_job_bureaucrat value = 0 }
    set_variable = { which = hsr_corp_total_job_entertainer value = 0 }
    set_variable = { which = hsr_corp_total_job_farmer value = 0 }
    set_variable = { which = hsr_corp_total_job_foundry value = 0 }
    set_variable = { which = hsr_corp_total_job_healthcare value = 0 }
    set_variable = { which = hsr_corp_total_job_miner value = 0 }
    set_variable = { which = hsr_corp_total_job_researcher value = 0 }
    set_variable = { which = hsr_corp_total_job_roboticist value = 0 }
    set_variable = { which = hsr_corp_total_job_soldier value = 0 }
    set_variable = { which = hsr_corp_total_job_technician value = 0 }
    set_variable = { which = hsr_corp_total_job_trader value = 0 }
    set_variable = { which = hsr_corp_industry_1 value = 0 }
    set_variable = { which = hsr_corp_industry_2 value = 0 }
    set_variable = { which = hsr_corp_industry_3 value = 0 }
    set_variable = { which = hsr_corp_industry_4 value = 0 }
    set_variable = { which = hsr_corp_industry_5 value = 0 }
    set_variable = { which = hsr_corp_industry_6 value = 0 }
    set_variable = { which = hsr_corp_industry_7 value = 0 }
    set_variable = { which = hsr_corp_industry_8 value = 0 }
    set_variable = { which = hsr_corp_industry_9 value = 0 }
    set_variable = { which = hsr_corp_s1_jobs value = 0 }
    if = {
        limit = {
            hsr_corp_slot_operating = { SLOT = 1 }
        }
        hsr_corp_tally_slot = { SLOT = 1 }
    }
    set_variable = { which = hsr_corp_s2_jobs value = 0 }
    if = {
        limit = {
            hsr_corp_slot_operating = { SLOT = 2 }
        }
        hsr_corp_tally_slot = { SLOT = 2 }
    }
    set_variable = { which = hsr_corp_s3_jobs value = 0 }
    if = {
        limit = {
            hsr_corp_slot_operating = { SLOT = 3 }
        }
        hsr_corp_tally_slot = { SLOT = 3 }
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_defense value > 25 }
        }
        set_variable = { which = hsr_corp_defense value = 25 }
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_occupied value = 0 }
        }
        remove_carrier_flag = hsr_corp_has_records
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_occupied value > 0 }
            NOT = {
                has_deposit = d_hsr_corp_enterprise
            }
        }
        add_deposit = d_hsr_corp_enterprise
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_occupied value = 0 }
            has_deposit = d_hsr_corp_enterprise
        }
        remove_deposit = d_hsr_corp_enterprise
    }
}

# 体制加成对公司总岗位取整一次；余数固定归主营，避免读档后随机变化。
hsr_corp_tally_slot = {
    optimize_memory
    change_variable = { which = hsr_corp_count value = 1 }
    change_variable = { which = hsr_corp_defense value = hsr_corp_s$SLOT$_level }
    set_variable = { which = hsr_corp_main_jobs value = hsr_corp_s$SLOT$_level }
    divide_variable = { which = hsr_corp_main_jobs value = 2 }
    ceiling_variable = hsr_corp_main_jobs
    set_variable = { which = hsr_corp_side_jobs value = hsr_corp_s$SLOT$_level }
    divide_variable = { which = hsr_corp_side_jobs value = 4 }
    floor_variable = hsr_corp_side_jobs
    set_variable = { which = hsr_corp_s$SLOT$_jobs value = hsr_corp_main_jobs }
    if = {
        limit = {
            check_variable = { which = hsr_corp_s$SLOT$_level value >= 4 }
            check_variable = { which = hsr_corp_s$SLOT$_secondary_1 value > 0 }
        }
        change_variable = { which = hsr_corp_s$SLOT$_jobs value = hsr_corp_side_jobs }
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_s$SLOT$_level value >= 7 }
            check_variable = { which = hsr_corp_s$SLOT$_secondary_2 value > 0 }
        }
        change_variable = { which = hsr_corp_s$SLOT$_jobs value = hsr_corp_side_jobs }
    }
    set_variable = { which = hsr_corp_extra_jobs value = 0 }
    if = {
        limit = {
            check_variable = { which = hsr_corp_s$SLOT$_regime value = 1 }
        }
        set_variable = { which = hsr_corp_extra_jobs value = hsr_corp_s$SLOT$_jobs }
        multiply_variable = { which = hsr_corp_extra_jobs value = 0.2 }
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_s$SLOT$_regime value = 2 }
        }
        set_variable = { which = hsr_corp_extra_jobs value = hsr_corp_s$SLOT$_jobs }
        multiply_variable = { which = hsr_corp_extra_jobs value = 0.5 }
    }
    floor_variable = hsr_corp_extra_jobs
    change_variable = { which = hsr_corp_main_jobs value = hsr_corp_extra_jobs }
    change_variable = { which = hsr_corp_s$SLOT$_jobs value = hsr_corp_extra_jobs }
    change_variable = { which = hsr_corp_total_jobs value = hsr_corp_s$SLOT$_jobs }
    set_variable = { which = hsr_corp_map_industry value = hsr_corp_s$SLOT$_primary }
    set_variable = { which = hsr_corp_map_jobs value = hsr_corp_main_jobs }
    hsr_corp_map_industry_jobs = yes

            check_variable = { which = hsr_corp_s$SLOT$_level value >= 4 }

        set_variable = { which = hsr_corp_map_jobs value = hsr_corp_side_jobs }
        hsr_corp_map_industry_jobs = yes
    }
    if = {
        limit = {
            check_variable = { which = hsr_corp_s$SLOT$_level value >= 7 }
            check_variable = { which = hsr_corp_s$SLOT$_secondary_2 value > 0 }
        }
        set_variable = { which = hsr_corp_map_industry value = hsr_corp_s$SLOT$_secondary_2 }
        set_variable = { which = hsr_corp_map_jobs value = hsr_corp_side_jobs }
        hsr_corp_map_industry_jobs = yes
    }
    if = {

            OR = {
