# 货币战争公司系统 v1.4；独立扩展文件，中文说明见 docs/货币战争。
# 状态：0空、1建设、2活动、3危机、4清算。未经完整游戏验收。
hsr_corp_valid_hq = {
    is_colony = yes
    exists = owner
}

hsr_corp_slot_empty = {
    NOT = {
        check_variable = { which = hsr_corp_s$SLOT$_state value > 0 }
    }
    NOT = {
        has_carrier_flag = hsr_corp_notice_s$SLOT$
    }
}

hsr_corp_slot_active = {
    check_variable = { which = hsr_corp_s$SLOT$_state value = 2 }
    check_variable = { which = hsr_corp_s$SLOT$_id value > 0 }
}

# 经营危机继续经营并保留岗位；交易与升级仍要求ACTIVE。
hsr_corp_slot_operating = {
    # HSR_SYSTEM_SWITCH_BEGIN
    NOT = {
        has_global_flag = hsr_currency_war_disabled
    }
    # HSR_SYSTEM_SWITCH_END
    OR = {
        check_variable = { which = hsr_corp_s$SLOT$_state value = 2 }
        check_variable = { which = hsr_corp_s$SLOT$_state value = 3 }
    }
    check_variable = { which = hsr_corp_s$SLOT$_id value > 0 }
}

hsr_corp_cooperative_allowed = {
    owner = {
        OR = {
            has_ethic = ethic_egalitarian
            has_ethic = ethic_fanatic_egalitarian
        }
    }
}

hsr_corp_session_valid = {
    # HSR_SYSTEM_SWITCH_BEGIN
    NOT = {
        has_global_flag = hsr_currency_war_disabled
    }
    # HSR_SYSTEM_SWITCH_END
    hsr_corp_valid_hq = yes
    exists = event_target:hsr_corp_actor
    check_variable = { which = hsr_corp_session_id value = event_target:hsr_corp_actor.hsr_corp_selected_id }
    OR = {
        AND = {
            check_variable = { which = hsr_corp_selected value = 1 }
            check_variable = { which = hsr_corp_s1_id value = hsr_corp_session_id }
            check_variable = { which = hsr_corp_s1_state value > 0 }
            NOT = {
                check_variable = { which = hsr_corp_s1_state value = 4 }
            }
        }
        AND = {
            check_variable = { which = hsr_corp_selected value = 2 }
            check_variable = { which = hsr_corp_s2_id value = hsr_corp_session_id }
            check_variable = { which = hsr_corp_s2_state value > 0 }
            NOT = {
                check_variable = { which = hsr_corp_s2_state value = 4 }
            }
        }
        AND = {
            check_variable = { which = hsr_corp_selected value = 3 }
            check_variable = { which = hsr_corp_s3_id value = hsr_corp_session_id }
            check_variable = { which = hsr_corp_s3_state value > 0 }
            NOT = {
                check_variable = { which = hsr_corp_s3_state value = 4 }
            }
        }
    }
}

hsr_corp_can_manage_slot = {
    hsr_corp_valid_hq = yes
    exists = event_target:hsr_corp_actor
    OR = {
        AND = {
            has_carrier_flag = hsr_corp_rights_suspended_s$SLOT$
            owner = {
                is_same_value = event_target:hsr_corp_actor
            }
        }
        AND = {
            NOT = {
                has_carrier_flag = hsr_corp_rights_suspended_s$SLOT$
            }
            OR = {
                AND = {
                    check_variable = { which = hsr_corp_s$SLOT$_foreign_control value = 1 }
                    exists = event_target:hsr_corp_foreign_s$SLOT$@this
                    event_target:hsr_corp_foreign_s$SLOT$@this = {
                        is_same_value = event_target:hsr_corp_actor
                    }
                }
                AND = {
                    NOT = {
                        check_variable = { which = hsr_corp_s$SLOT$_foreign_control value = 1 }
                    }
                    OR = {
                        AND = {
                            exists = event_target:hsr_corp_founder_s$SLOT$@this
                            event_target:hsr_corp_founder_s$SLOT$@this = {
                                is_same_value = event_target:hsr_corp_actor
                            }
                        }
                        AND = {
                            NOT = {
                                exists = event_target:hsr_corp_founder_s$SLOT$@this
                            }
                            owner = {
                                is_same_value = event_target:hsr_corp_actor
                            }
                        }
                    }
                }
            }
        }
    }
}

hsr_corp_can_visit = {
    # HSR_SYSTEM_SWITCH_BEGIN
    NOT = {
        has_global_flag = hsr_currency_war_disabled
    }
    # HSR_SYSTEM_SWITCH_END
    hsr_corp_valid_hq = yes
    exists = event_target:hsr_corp_actor
    OR = {
        owner = {
            is_same_value = event_target:hsr_corp_actor
        }
        AND = {
            event_target:hsr_corp_actor = {
                has_communications = prev.owner
                NOT = {
                    is_at_war_with = prev.owner
                }
            }
            owner = {
                is_country_type = default
            }
        }
    }
}

# 控制国作用域：防御战争或国境内实际出现敌对天灾舰队。只在申请/月度解锁时查询。
hsr_corp_defensive_emergency = {
    OR = {
        any_war = {
            any_defender = {
                is_same_value = event_target:hsr_corp_controller
            }
        }
        any_system_within_border = {
            any_fleet_in_system = {
                exists = owner
                owner = {
                    is_crisis_faction = yes
                    is_hostile = event_target:hsr_corp_controller
                }
            }
        }
    }
}

hsr_corp_creation_allowed = {
    # HSR_SYSTEM_SWITCH_BEGIN
    NOT = {
        has_global_flag = hsr_currency_war_disabled
    }
    # HSR_SYSTEM_SWITCH_END
    hsr_corp_valid_hq = yes
    hsr_corp_slot_empty = { SLOT = $SLOT$ }
    NOT = {
        has_carrier_flag = hsr_corp_notice_s$SLOT$
    }
    exists = event_target:hsr_corp_actor
    owner = {
        is_same_value = event_target:hsr_corp_actor
    }
    exists = event_target:global_event_country
    OR = {
        check_variable = { which = hsr_corp_setup_level value = 1 }
        check_variable = { which = hsr_corp_setup_level value = 4 }
        check_variable = { which = hsr_corp_setup_level value = 7 }
    }
    check_variable = { which = hsr_corp_setup_primary value >= 1 }
    check_variable = { which = hsr_corp_setup_primary value <= 9 }
    check_variable = { which = hsr_corp_setup_regime value >= 1 }
    check_variable = { which = hsr_corp_setup_regime value <= 4 }
    OR = {
        NOT = {
            check_variable = { which = hsr_corp_setup_regime value = 2 }
        }
        hsr_corp_cooperative_allowed = yes
    }
    OR = {
        AND = {
            check_variable = { which = hsr_corp_setup_payment value = 0 }
            owner = {
                has_resource = { type = trade amount >= prev.hsr_corp_quote }
            }
        }
        AND = {
            check_variable = { which = hsr_corp_setup_payment value = 1 }
            owner = {
                has_ascension_perk = ap_hsr_corp_currency_war
                check_variable = { which = hsr_credits_country_trade_storage value >= prev.hsr_corp_quote_national }
            }
            check_variable = { which = hsr_credits_planet_trade_storage value >= hsr_corp_quote_local }
        }
    }
}

hsr_corp_market_allowed = {
    hsr_corp_slot_active = { SLOT = $SLOT$ }
    hsr_corp_can_visit = yes
    check_variable = { which = hsr_corp_s$SLOT$_listed value = 1 }
    check_variable = { which = hsr_corp_s$SLOT$_pool_shares value = value:hsr_corp_expected_pool_s$SLOT$ }
    OR = {
        check_variable = { which = hsr_corp_order_qty value = 1 }
        check_variable = { which = hsr_corp_order_qty value = 5 }
        check_variable = { which = hsr_corp_order_qty value = 10 }
    }
    event_target:hsr_corp_actor = { NOT = { has_country_flag = hsr_corp_trade_s$SLOT$@event_target:hsr_corp_hq } }
    OR = {
        AND = {
            exists = event_target:hsr_corp_founder_s$SLOT$@this
            event_target:hsr_corp_founder_s$SLOT$@this = { is_same_value = event_target:hsr_corp_actor }
        }
        AND = {
            NOT = { exists = event_target:hsr_corp_founder_s$SLOT$@this }
            owner = { is_same_value = event_target:hsr_corp_actor }
        }
        NOT = { exists = event_target:hsr_corp_foreign_s$SLOT$@this }
        AND = {
            exists = event_target:hsr_corp_foreign_s$SLOT$@this
            event_target:hsr_corp_foreign_s$SLOT$@this = { is_same_value = event_target:hsr_corp_actor }
        }
    }
}


# 操作条件使用普通 scripted_trigger 定义；按钮与执行入口分别核验当前真实状态。
hsr_corp_upgrade_allowed = {
    hsr_corp_can_manage_slot = { SLOT = $SLOT$ }
    hsr_corp_slot_active = { SLOT = $SLOT$ }
    check_variable = { which = hsr_corp_s$SLOT$_level value < 10 }
    check_variable = { which = hsr_corp_s$SLOT$_cash value >= hsr_corp_upgrade_cost }
    NOT = { has_carrier_flag = hsr_corp_upgrade_s$SLOT$ }
}

hsr_corp_expand_allowed = {
    hsr_corp_can_manage_slot = { SLOT = $SLOT$ }
    hsr_corp_slot_active = { SLOT = $SLOT$ }
    check_variable = { which = hsr_corp_choose_industry value >= 1 }
    check_variable = { which = hsr_corp_choose_industry value <= 9 }
    NOT = { check_variable = { which = hsr_corp_s$SLOT$_primary value = hsr_corp_choose_industry } }
    NOT = { check_variable = { which = hsr_corp_s$SLOT$_secondary_1 value = hsr_corp_choose_industry } }
    NOT = { check_variable = { which = hsr_corp_s$SLOT$_secondary_2 value = hsr_corp_choose_industry } }
    check_variable = { which = hsr_corp_s$SLOT$_level value >= $LEVEL$ }
    check_variable = { which = hsr_corp_s$SLOT$_secondary_$SIDE$ value = 0 }
    check_variable = { which = hsr_corp_s$SLOT$_cash value >= hsr_corp_side_$SIDE$_cost }
}

hsr_corp_buy_allowed = {
    hsr_corp_market_allowed = { SLOT = $SLOT$ }
    check_variable = { which = hsr_corp_s$SLOT$_pool_shares value >= hsr_corp_order_qty }
    event_target:hsr_corp_actor = { has_resource = { type = trade amount >= prev.hsr_corp_order_total } }
}

hsr_corp_sell_allowed = {
    hsr_corp_market_allowed = { SLOT = $SLOT$ }
    check_variable = { which = hsr_corp_s$SLOT$_market_reserve value >= hsr_corp_order_gross }
    OR = {
        AND = {
            exists = event_target:hsr_corp_founder_s$SLOT$@this
            event_target:hsr_corp_founder_s$SLOT$@this = { is_same_value = event_target:hsr_corp_actor }
            check_variable = { which = hsr_corp_s$SLOT$_founder_shares value >= hsr_corp_order_qty }
        }
        AND = {
            NOT = { exists = event_target:hsr_corp_founder_s$SLOT$@this }
            owner = { is_same_value = event_target:hsr_corp_actor }
            check_variable = { which = hsr_corp_s$SLOT$_founder_shares value >= hsr_corp_order_qty }
        }
        AND = {
            exists = event_target:hsr_corp_foreign_s$SLOT$@this
            event_target:hsr_corp_foreign_s$SLOT$@this = { is_same_value = event_target:hsr_corp_actor }

    }
}

# 总部载体作用域；仅统计仍存在且实际持有股份的外部国家。
hsr_corp_outside_shareholder = {
    exists = event_target:hsr_corp_$ROLE$_s$SLOT$@this
    check_variable = { which = hsr_corp_s$SLOT$_$ROLE$_shares value > 0 }
    event_target:hsr_corp_$ROLE$_s$SLOT$@this = {
        NOT = { is_same_value = prev.owner }
    }
}

hsr_corp_controller_outside = {
    NOT = { has_carrier_flag = hsr_corp_rights_suspended_s$SLOT$ }
    OR = {
        AND = {
            check_variable = { which = hsr_corp_s$SLOT$_foreign_control value = 1 }
            exists = event_target:hsr_corp_foreign_s$SLOT$@this
            event_target:hsr_corp_foreign_s$SLOT$@this = { NOT = { is_same_value = prev.owner } }
        }
        AND = {
            NOT = {
                check_variable = { which = hsr_corp_s$SLOT$_foreign_control value = 1 }
                exists = event_target:hsr_corp_foreign_s$SLOT$@this
            }
            exists = event_target:hsr_corp_founder_s$SLOT$@this
            event_target:hsr_corp_founder_s$SLOT$@this = { NOT = { is_same_value = prev.owner } }
        }
    }
}

hsr_corp_illegal_allowed = {
    hsr_corp_slot_operating = { SLOT = $SLOT$ }
    owner = { is_same_value = event_target:hsr_corp_actor }
    NOT = { has_carrier_flag = hsr_corp_illegal_cd_s$SLOT$ }
    check_variable = { which = hsr_corp_s$SLOT$_offer_escrow value = 0 }
    check_variable = { which = hsr_corp_illegal_kind value >= 1 }
    check_variable = { which = hsr_corp_illegal_kind value <= 4 }
    OR = {
        hsr_corp_outside_shareholder = { SLOT = $SLOT$ ROLE = founder }
        hsr_corp_outside_shareholder = { SLOT = $SLOT$ ROLE = foreign }
    }
    OR = {
        NOT = { check_variable = { which = hsr_corp_illegal_kind value = 1 } }
        check_variable = { which = hsr_corp_s$SLOT$_cash value > 0 }
    }
    OR = {
        NOT = { check_variable = { which = hsr_corp_illegal_kind value = 3 } }
        hsr_corp_controller_outside = { SLOT = $SLOT$ }
    }
}
