	allow = {
		custom_tooltip = {
			fail_text = market_confirm_1_effect_disabled
			hidden_trigger = {
				from = {
					resource_stockpile_compare = {
						resource = sr_dark_matter
						value >= 50
					}
				}
			}
		}
	}
	potential = {
		from = {
			check_variable = {
				which = xianzhou_market_page
				value = 1
			}
		}
	}
	effect = {
		hidden_effect = {
			from = {

				}
			}
		}
	}
