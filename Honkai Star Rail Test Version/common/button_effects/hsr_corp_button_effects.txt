        # HSR_FINANCE_TOOLTIP：隐藏内部标记、作用域及变量，只显示本地化条件。

            fail_text = hsr_finance_requirements
            hidden_trigger = {
                # HSR_SYSTEM_SWITCH_BEGIN
                NOT = {
                    has_global_flag = hsr_credit_system_disabled
                }
                # HSR_SYSTEM_SWITCH_END
                from = {
                    custom_tooltip = {
                        fail_text = hsr_finance_blocked
                        hidden_trigger = {
                            has_policy_flag = allow_use_eco
                            NOT = {
                                has_country_flag = hsr_finance_overdue
                            }
                            check_variable = { which = hsr_finance_remaining value >= value:hsr_finance_loan_small }
                        }
                    }
                }
            }
        }
    }
    effect = {
        from = {
            hidden_effect = {
                from = {
                    save_event_target_as = hsr_corp_actor
                }
                hsr_finance_borrow = { AMOUNT = value:hsr_finance_loan_small }
            }
        }
    }
}
