hsr_stage_play_1_start_effect = {
    allow = {
         custom_tooltip = {
            fail_text = has_choose
            hidden_trigger = {
                from = {
                    not = {has_country_flag = hsr_stage_play_1_choose}
                    has_country_flag = hsr_stage_play_0_set
                    has_country_flag = haimeixiewan
                    
                }
            }
        }
        
    }
    potential = {
        always = yes
    }
    effect = {
        set_country_flag = hsr_stage_play_1_choose
        remove_country_flag = hsr_stage_play_0_set
        set_variable = {
            which = hsr_stage_play_0
            value = 1
        }
    }
}
hsr_stage_play_2_start_effect = {
    allow = {
         custom_tooltip = {
            fail_text = has_choose
            hidden_trigger = {
                from = {
                    not = {has_country_flag = hsr_stage_play_2_choose}
                    has_country_flag = hsr_stage_play_0_set
                    
                }
            }
        }
        
    }
    potential = {
        always = yes
    }
    effect = {
        set_country_flag = hsr_stage_play_2_choose
        remove_country_flag = hsr_stage_play_0_set
        set_variable = {
            which = hsr_stage_play_0
            value = 2
        }
        set_variable = {which = hsr_stage_play_2_stage_1 value = 0}
        set_variable = {which = hsr_stage_play_2_stage_2 value = 0}
    }
}
